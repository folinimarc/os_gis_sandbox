# =========================
# CORE VECTOR (GeoPandas stack)
# =========================
geopandas        # high-level geospatial dataframe library
shapely          # geometry engine (GEOS, Shapely 2 backend)
pyogrio          # fast vector I/O (default engine in GeoPandas)
pyproj           # coordinate reference systems (PROJ)
pandas
numpy

# Optional low-level / legacy vector access
fiona            # alternative I/O engine (OGR-based, legacy but still useful)
gdal             # low-level bindings (osgeo.*), advanced use cases
    
# =========================
# RASTER
# =========================
rasterio         # raster I/O and processing (GDAL-based)
xarray           # N-dimensional arrays
rioxarray        # geospatial extension for xarray (CRS, clipping, etc.)

# =========================
# SPATIAL ANALYSIS
# =========================
pysal            # spatial statistics & econometrics
rasterstats      # zonal statistics (vector / raster bridge)

# =========================
# DATABASES / STORAGE
# =========================
pyarrow          # parquet / feather support
sqlalchemy       # required for database connections (GeoPandas uses it)
geoalchemy2      # spatial SQLAlchemy extension (PostGIS)
psycopg[binary]  # PostgreSQL driver (this is psycopg3, the modern driver)
psycopg2-binary  # alternative legacy driver

# =========================
# STAC / CLOUD GEOSPATIAL
# =========================
pystac           # STAC spec implementation (catalogs, items, assets)
pystac-client    # query STAC APIs (search catalogs like Planetary Computer)
stackstac        # convert STAC items to xarray (analysis-ready data cubes)

# =========================
# PLOTTING / VISUALIZATION
# =========================
matplotlib       # static plotting (GeoDataFrame.plot)
mapclassify      # choropleth classification schemes
folium           # leaflet maps (GeoDataFrame.explore)
ipyleaflet       # interactive maps in Jupyter
contextily       # basemaps (web tiles for matplotlib plots)

# =========================
# SPECIALIZED
# =========================
osmnx            # OpenStreetMap + network analysis
movingpandas     # trajectory / movement data
momepy           # urban morphology analysis

# =========================
# WEB SERVICES
# =========================
OWSLib           # OGC services (WMS, WFS, etc.)

# =========================
# NOTEBOOK / DEV ENV
# =========================
jupyterlab              # interactive environment
jupyter-lsp             # language server integration
python-lsp-server[all]  # code intelligence

# =========================
# DATA / TEACHING UTILITIES
# =========================
geodatasets      # small sample datasets (used in GeoPandas docs)

# =========================
# OFFICE / FILE FORMATS
# =========================
openpyxl         # Excel (.xlsx) read/write
xlsxwriter       # Excel writing engine
xlrd             # ONLY for legacy .xls (not .xlsx)
